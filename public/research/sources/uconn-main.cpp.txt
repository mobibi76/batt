// 2) Helpers

double linearInterpolate(double x, const vector<double>& xp, const vector<double>& fp) {
  if (xp.empty() || fp.empty()) return 0.0;
  if (x <= xp.front()) return fp.front();
  if (x >= xp.back())  return fp.back();
  auto it = std::lower_bound(xp.begin(), xp.end(), x);
  if (it == xp.begin()) return fp.front();
  if (it == xp.end())   return fp.back();
  size_t i = size_t((it - xp.begin()) - 1);
  double x0 = xp[i], x1 = xp[i + 1];
  double y0 = fp[i], y1 = fp[i + 1];
  double r  = (x - x0) / (x1 - x0);
  return y0 + r * (y1 - y0);
}

double u_sig(double t, const vector<double>& tVector, const vector<double>& iVector) {
  return linearInterpolate(t, tVector, iVector);
}

double VocLUT(double soc, const Cell& cell) {
  return linearInterpolate(soc, cell.SOC, cell.OCV);
}

StatFeatures computeVoltStats(const vector<double>& V) {
  StatFeatures s{};
  if (V.empty()) {
    s.volt_mean = s.volt_max = s.volt_min =
    s.volt_var  = s.volt_skew = s.volt_kurtosis = std::numeric_limits<double>::quiet_NaN();
    return s;
  }

  const size_t n = V.size();

  // mean = np.mean(v)
  double mean = std::accumulate(V.begin(), V.end(), 0.0) / static_cast<double>(n);

  // central moments: m2, m3, m4 (population, divide by n)
  double m2 = 0.0, m3 = 0.0, m4 = 0.0;
  for (double v : V) {
    double d  = v - mean;
    double d2 = d * d;
    m2 += d2;
    m3 += d2 * d;
    m4 += d2 * d2;
  }
  m2 /= static_cast<double>(n);
  m3 /= static_cast<double>(n);
  m4 /= static_cast<double>(n);

  double var   = m2;   // np.var(v) with ddof=0
  double skew  = 0.0;
  double kurt  = 0.0;

  if (var > 0.0) {
    double sigma = std::sqrt(var);
    // scipy.stats.skew(v, bias=True) → m3 / sigma^3
    skew = m3 / (sigma * sigma * sigma);
    // scipy.stats.kurtosis(v, fisher=True, bias=True) → m4/m2^2 - 3
    kurt = m4 / (var * var) - 3.0;
  }

  s.volt_mean     = mean;
  s.volt_max      = *std::max_element(V.begin(), V.end());
  s.volt_min      = *std::min_element(V.begin(), V.end());
  s.volt_var      = var;
  s.volt_skew     = skew;
  s.volt_kurtosis = kurt;

  return s;
}

static const size_t LFP_OCV_TABLE_SIZE = sizeof(LFP_OCV_TABLE) / sizeof(LFP_OCV_TABLE[0]);

Cell loadLfpCell() {
  Cell cell;
  cell.OCV.reserve(LFP_OCV_TABLE_SIZE);
  cell.SOC.reserve(LFP_OCV_TABLE_SIZE);
  for (size_t i = 0; i < LFP_OCV_TABLE_SIZE; ++i) {
    cell.OCV.push_back(LFP_OCV_TABLE[i].voltage);
    cell.SOC.push_back(LFP_OCV_TABLE[i].soc);
  }
  cell.Capacity = 1.2; // Ah
  return cell;
}

//==============================================================================
// 4) State-space ECM simulation (Euler dt=1s)
//==============================================================================
void cellModelEvalStateSpace(int n,
                             double R0,
                             const vector<double>& RArray,
                             const vector<double>& CArray,
                             const Cell& cell,
                             const vector<double>& iVector,
                             const vector<double>& tVector,
                             const vector<double>& Vt_full,
                             vector<double>& soc_out,
                             vector<double>& Vt_out) {
  const double dt = 1.0;
  const double t_end = tVector.back();
  const int stateSize = n + 1;  // n RC states + SOC

  vector<double> x(stateSize, 0.0);
  // initial SOC via OCV inversion of first measured V
  x[stateSize - 1] = linearInterpolate(Vt_full[0], cell.OCV, cell.SOC);

  soc_out.clear(); Vt_out.clear();
  soc_out.push_back(x[stateSize - 1]);

  double sum_RC=0.0;
  for (int i=0;i<n;++i) sum_RC += x[i];
  double V0 = sum_RC + VocLUT(x[stateSize - 1], cell) + R0 * u_sig(tVector[0], tVector, iVector);
  Vt_out.push_back(V0);

  double t = tVector[0];
  size_t sample_index = 1;

  while (t < t_end) {
    double u = u_sig(t, tVector, iVector);
    vector<double> dx(stateSize, 0.0);

    for (int i=0;i<n;++i) {
      double tau = RArray[i]*CArray[i];
      dx[i] = -x[i]/tau + (1.0/CArray[i])*u;
    }
    dx[stateSize-1] = (1.0 / (cell.Capacity * 3600.0)) * u;  // SOC integrator

    for (int i=0;i<stateSize;++i) x[i] += dt*dx[i];
    t += dt;

    if (sample_index < tVector.size() && t >= tVector[sample_index]) {
      soc_out.push_back(x[stateSize - 1]);
      sum_RC=0.0; for (int i=0;i<n;++i) sum_RC += x[i];
      double Vt = sum_RC + VocLUT(x[stateSize - 1], cell)
                + R0 * u_sig(tVector[sample_index], tVector, iVector);
      Vt_out.push_back(Vt);
      ++sample_index;
    }
  }
}

//==============================================================================
// 5) Objective + tiny coordinate-descent optimizer
//==============================================================================
double ECMOptimization(const vector<double>& x, void* data) {
  OptimData* opt = static_cast<OptimData*>(data);
  int n = opt->n;
  vector<double> RArray(x.begin(), x.begin()+n);
  vector<double> CArray(x.begin()+n, x.begin()+2*n);
  const double fixed_R0 = 0.01;  // fixed during fitting

  vector<double> soc_sim, V_sim;
  cellModelEvalStateSpace(n, fixed_R0, RArray, CArray,
                          opt->cell, opt->iVector, opt->tVector, opt->Vt_full,
                          soc_sim, V_sim);

  double err=0.0;
  size_t N = std::min(V_sim.size(), opt->Vt_ref.size());
  for (size_t i=0;i<N;++i) {
    double d = V_sim[i] - opt->Vt_ref[i];
    err += d*d;
  }
  return err;
}

vector<vector<double>> generate_initial_guesses(int n_RC, int num=20, unsigned int seed=42) {
  std::mt19937 rng(seed);
  std::normal_distribution<double> normR(0.0, 0.02);
  std::normal_distribution<double> normC(0.0, 200.0);
  vector<vector<double>> guesses; guesses.reserve(num);

  vector<double> Rb(n_RC), Cb(n_RC);
  for (int i=0;i<n_RC;++i) {
    Rb[i] = 0.02 + (0.2 - 0.02) * (double(i) / double(std::max(1, n_RC-1)));
    Cb[i] = 100.0 + (2000.0 - 100.0) * (double(i) / double(std::max(1, n_RC-1)));
  }

  for (int g=0; g<num; ++g) {
    vector<double> x(2*n_RC);
    for (int k=0;k<n_RC;++k) {
      double Rg = std::clamp(Rb[k] + normR(rng), 1e-3, 0.9);
      double Cg = std::clamp(Cb[k] + normC(rng), 10.0, 1e4);
      x[k] = Rg; x[n_RC+k] = Cg;
    }
    guesses.push_back(x);
  }
  return guesses;
}

void clamp_params(vector<double>& x, int n_RC) {
  for (int i=0;i<n_RC;++i) x[i]          = std::clamp(x[i], 1e-3, 0.9);
  for (int i=0;i<n_RC;++i) x[n_RC+i]     = std::clamp(x[n_RC+i], 10.0, 1e4);
}

vector<double> lightweightOptimizer(const vector<double>& x0,
                                    double (*obj)(const vector<double>&, void*),
                                    void* data, int n_RC,
                                    double tol=1e-4, int maxIter=400) {
  vector<double> x=x0; clamp_params(x,n_RC);
  int n=int(x.size());
  vector<double> step(n);
  for (int i=0;i<n;++i) step[i] = (fabs(x[i])>0 ? fabs(x[i])*0.1 : 0.01);

  double bestVal = obj(x,data);
  for (int it=0; it<maxIter; ++it) {
    bool improved=false;
    for (int i=0;i<n;++i) {
      vector<double> xNew=x; xNew[i]+=step[i]; clamp_params(xNew,n_RC);
      double v = obj(xNew,data);
      if (v<bestVal) { x.swap(xNew); bestVal=v; improved=true; continue; }
      xNew=x; xNew[i]-=step[i]; clamp_params(xNew,n_RC);
      v = obj(xNew,data);
      if (v<bestVal) { x.swap(xNew); bestVal=v; improved=true; }
    }
    if (!improved) {
      bool small=true;
      for (int i=0;i<n;++i) { step[i]*=0.7; if (step[i]>tol) small=false; }
      if (small) break;
    }
  }
  return x;
}

//==============================================================================
// 6) Top-k ensemble + SOC soft gating (no JSON)
//==============================================================================
struct BandExpert {
  vector<int>    feature_indices;
  vector<double> scaler_mean;
  vector<double> scaler_scale;
  vector<double> coef;
  double         intercept;
  double         band_lo_pct;  // inclusive (%SOC)
  double         band_hi_pct;  // exclusive (%SOC)
};

struct TopkSocGatedEnsembleLFP {
  const char* feature_order[12];
  vector<BandExpert> bands;

  static double eval_expert(const BandExpert& b, const vector<double>& x) {
    double y=b.intercept;
    size_t k=b.feature_indices.size();
    for (size_t j=0;j<k;++j) {
      int idx=b.feature_indices[j];
      double z = (fabs(b.scaler_scale[j])>1e-12)
                 ? (x[idx]-b.scaler_mean[j])/b.scaler_scale[j]
                 : (x[idx]-b.scaler_mean[j]);
      y += b.coef[j]*z;
    }
    return y;
  }

  static double tri_weight(double soc_pct, double lo, double hi) {
    if (!(hi > lo)) return 0.0;
    if (soc_pct <= lo || soc_pct >= hi) return 0.0;
    double center = 0.5*(lo+hi);
    double half   = 0.5*(hi-lo);
    return std::max(0.0, 1.0 - fabs(soc_pct - center)/half);
  }

  double predict_gated(const vector<double>& x, double soc_pct, vector<double>* out_w=nullptr) const {
    if (bands.empty()) return 0.0;
    vector<double> w(bands.size(), 0.0);
    for (size_t k=0;k<bands.size();++k) w[k] = tri_weight(soc_pct, bands[k].band_lo_pct, bands[k].band_hi_pct);
    double wsum = std::accumulate(w.begin(), w.end(), 0.0);

    double y=0.0;
    if (wsum <= 1e-12) {
      for (const auto& b : bands) y += eval_expert(b, x);
      y /= double(bands.size());
      if (out_w) out_w->assign(bands.size(), 1.0/double(bands.size()));
      return y;
    }
    for (size_t k=0;k<bands.size();++k) y += (w[k]/wsum) * eval_expert(bands[k], x);
    if (out_w) { for (double& wk : w) wk /= wsum; *out_w = w; }
    return y;
  }

  double predict_uniform(const vector<double>& x) const {
    if (bands.empty()) return 0.0;
    double sum=0.0; for (const auto& b : bands) sum += eval_expert(b, x);
    return sum / double(bands.size());
  }
};

//==============================================================================
// 7) One-trace pipeline (expects exactly 100 samples) + SOC gating
//==============================================================================

// Returns free heap bytes via mallinfo (static RAM = data+bss is fixed at link time).
static uint32_t freeRamBytes() {
  struct mallinfo mi = mallinfo();
  return (uint32_t)mi.fordblks;
}

void run_one_trace(const vector<double>& Vt_full) {
  const uint32_t t_total_start = micros();

  if (Vt_full.size()!=100) { Serial.println("ERROR: need 100 samples."); return; }
  if (Vt_full.size()<=31)  { Serial.println("ERROR: step indices missing."); return; }

  const int L=100;
  Cell cell = loadLfpCell();

  // Active current profile; see ACTIVE_PULSE_PROFILE_NAME at top of file.
  const double C_nom = cell.Capacity;
  const int n_C5=ACTIVE_N_C5, n_C=ACTIVE_N_1C, n_zero=L-(n_C5+n_C);

  vector<double> iVector; iVector.reserve(L);
  for (int i=0;i<n_C5;++i) iVector.push_back(-C_nom/5.0);
  for (int i=0;i<n_C; ++i) iVector.push_back(-C_nom);
  for (int i=0;i<n_zero;++i) iVector.push_back(0.0);

  vector<double> tVector; tVector.reserve(L);
  for (int i=1;i<=L;++i) tVector.push_back(double(i));

  OptimData opt;
  opt.n=2; opt.cell=cell;
  opt.iVector=iVector; opt.tVector=tVector;
  opt.Vt_full=Vt_full; opt.Vt_ref=Vt_full;

  // --- Optimize (multi-start coordinate descent)
  const uint32_t t_ecm_start = micros();
  auto seeds = generate_initial_guesses(opt.n, 20, 42);
  double best_loss = std::numeric_limits<double>::infinity();
  vector<double> best_x;

  for (const auto& s : seeds) {
    vector<double> x = lightweightOptimizer(s, ECMOptimization, &opt, opt.n, 1e-4, 400);
    double v = ECMOptimization(x, &opt);
    if (v<best_loss) { best_loss=v; best_x=x; }
  }
  const uint32_t t_ecm_done = micros();
  if (best_x.empty()) { Serial.println("ERROR: optimizer failed."); return; }

  vector<double> R_opt(best_x.begin(), best_x.begin()+opt.n);
  vector<double> C_opt(best_x.begin()+opt.n, best_x.begin()+2*opt.n);

  // R0 from step 30→31: step current = C_nom - C_nom/5 = 0.8*C_nom
  double dV = Vt_full[32] - Vt_full[31];
  double R0_manual = (C_nom!=0.0)? fabs(dV)/(0.8*fabs(C_nom)) : 0.01;

  // Final sim uses R0_sim = R0_manual * 0.25 (matches Python r0_post_scale_lfp=0.25
  // used when generating the training features CSV; keeps residual_L1 in-distribution)
  const double R0_sim = R0_manual * 0.25;
  vector<double> SOC_ECM, Vt_fit;
  cellModelEvalStateSpace(opt.n, R0_sim, R_opt, C_opt,
                          opt.cell, opt.iVector, opt.tVector, opt.Vt_full,
                          SOC_ECM, Vt_fit);

  // Residual L1 like CSV
  double residual_L1=0.0;
  size_t N = std::min(Vt_fit.size(), opt.Vt_ref.size());
  for (size_t i=0;i<N;++i) residual_L1 += fabs(opt.Vt_ref[i]-Vt_fit[i]);

  // Voltage stats of measured trace
  StatFeatures st = computeVoltStats(Vt_full);

  // Feature vector order must match ensemble
  vector<double> feat(12);
  feat[0]=R_opt[0]; feat[1]=R_opt[1]; feat[2]=C_opt[0]; feat[3]=C_opt[1];
  feat[4]=R0_manual; feat[5]=residual_L1; feat[6]=st.volt_mean; feat[7]=st.volt_max;
  feat[8]=st.volt_min;  feat[9]=st.volt_var; feat[10]=st.volt_skew; feat[11]=st.volt_kurtosis;

  auto ens = buildLfpEnsemble();

  // ---- Raw predictions from ensemble
  const uint32_t t_infer_start = micros();
  double soh_uniform_raw = ens.predict_uniform(feat);

  // ---- SOC-gated prediction using recalibrated SOC (only used for gating)
  double soc_ecm_last = (!SOC_ECM.empty() ? SOC_ECM.back() : 0.5); // 0..1
  double soc_pct_raw  = 100.0 * std::clamp(soc_ecm_last, 0.0, 1.0); // ECM %
  double soc_pct_cal  = std::clamp(SOC_CAL_M*soc_pct_raw + SOC_CAL_B, 5.0, 95.0);

  vector<double> w_norm;
  double soh_gated_raw = ens.predict_gated(feat, soc_pct_cal, &w_norm);
  const uint32_t t_infer_done = micros();

  // Optional *output* calibration (compile-time switch)
  double soh_uniform = apply_soh_output_cal(soh_uniform_raw);
  double soh_gated   = apply_soh_output_cal(soh_gated_raw);


  // DEBUG line for Python checker:
  // DEBUG,volt_mean,volt_max,volt_min,volt_var,volt_skew,volt_kurtosis,soc_pct_raw,soc_pct_cal
  Serial.print("DEBUG,");
  Serial.print(st.volt_mean,10);      Serial.print(",");
  Serial.print(st.volt_max,10);       Serial.print(",");
  Serial.print(st.volt_min,10);       Serial.print(",");
  Serial.print(st.volt_var,10);       Serial.print(",");
  Serial.print(st.volt_skew,10);      Serial.print(",");
  Serial.print(st.volt_kurtosis,10);  Serial.print(",");
  Serial.print(soc_pct_raw,3);        Serial.print(",");
  Serial.print(soc_pct_cal,3);
  Serial.println();

  // // Also print the SOC gating weights for visibility
  // vector<double> w_norm;
  // double soh_gated = ens.predict_gated(feat, soc_pct_cal, &w_norm);

  // GATE,<ECM_SOC%>,<CAL_SOC%>,w_band1,w_band2,w_band3
  Serial.print("GATE,");
  Serial.print(soc_pct_raw, 3);  Serial.print(",");
  Serial.print(soc_pct_cal, 3);  Serial.print(",");
  for (size_t k=0;k<w_norm.size();++k) {
    Serial.print(w_norm[k], 6);
    if (k+1<w_norm.size()) Serial.print(",");
  }
  Serial.println();

  // RESULT line (CSV-like): uniform and gated SOH, then a few diagnostics
  Serial.print("RESULT,");
  Serial.print(soh_uniform,7); Serial.print(",");
  Serial.print(soh_gated,7);   Serial.print(",");
  Serial.print(R_opt[0],7);    Serial.print(",");
  Serial.print(R_opt[1],7);    Serial.print(",");
  Serial.print(C_opt[0],7);    Serial.print(",");
  Serial.print(C_opt[1],7);    Serial.print(",");
  Serial.print(R0_manual,7);   Serial.print(",");
  Serial.print(residual_L1,7); Serial.print(",");
  Serial.print(st.volt_mean,7);Serial.print(",");
  Serial.print(st.volt_var,7);
  Serial.println();

  // PERF line: ECM fitting, ensemble inference, total (all in microseconds), free heap bytes
  // Format: PERF,<ecm_us>,<infer_us>,<total_us>,<free_heap_bytes>
  const uint32_t t_total_us  = micros() - t_total_start;
  const uint32_t t_ecm_us    = t_ecm_done - t_ecm_start;
  const uint32_t t_infer_us  = t_infer_done - t_infer_start;
  const uint32_t free_heap   = freeRamBytes();
  Serial.print("PERF,");
  Serial.print(t_ecm_us);   Serial.print(",");   // ECM fitting (us)
  Serial.print(t_infer_us); Serial.print(",");   // Ensemble inference (us)
  Serial.print(t_total_us); Serial.print(",");   // Total pipeline (us)
  Serial.print(free_heap);                        // Free heap (bytes)
  Serial.println();

}

//==============================================================================
// 8) Serial protocol: READY + PING + fixed 100 floats
//==============================================================================
void setup() {
  Serial.begin(115200);
  unsigned long t0 = millis();
  while (!Serial && (millis() - t0 < 3000)) { /* spin */ }
  Serial.setTimeout(2000);
  Serial.print("CONFIG,pulse_profile,");
  Serial.print(ACTIVE_PULSE_PROFILE_NAME);
  Serial.print(",n_C5,");
  Serial.print(ACTIVE_N_C5);
  Serial.print(",n_1C,");
  Serial.println(ACTIVE_N_1C);
  Serial.print("CONFIG,soh_calibration,");
  Serial.print(ACTIVE_SOH_CAL_NAME);
  Serial.print(",enabled,");
  Serial.print(USE_SOH_OUTPUT_CAL);
  Serial.print(",m,");
  Serial.print(SOH_CAL_M, 6);
  Serial.print(",b,");
  Serial.println(SOH_CAL_B, 6);
  Serial.println("READY");
}

static bool got_any_command = false;
static unsigned long last_ready_ms = 0;

void loop() {
  if (!got_any_command && millis() - last_ready_ms > 500) {
    Serial.println("READY");
    last_ready_ms = millis();
  }

  if (Serial.available()) {
    int c = Serial.peek();

    if (c == 'P' || c == 'p') {
      String cmd = Serial.readStringUntil('\n');
      cmd.trim();
      if (cmd.equalsIgnoreCase("PING")) {
        got_any_command = true;
        Serial.println("READY");
        return;
      }
    }

    vector<double> Vt_full; Vt_full.reserve(100);
    for (int i = 0; i < 100; ++i) {
      double v = Serial.parseFloat();
      Vt_full.push_back(v);
    }
    // Clear rest of the line (newline/comma noise) to keep protocol aligned
    while (Serial.available()) { char ch = Serial.peek(); if (ch=='\n') { Serial.read(); break; } Serial.read(); }

    got_any_command = true;
    run_one_trace(Vt_full);
    Serial.println("DONE");
  }
}
